cmake_minimum_required(VERSION 3.16)
project(dual_airy_deploy_cpp VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
# 共享核心包含外参读取、26 字节点数据、融合缓存和启动时差估计；无需 PCL。
add_library(dual_airy_core src/core.cpp)
target_include_directories(dual_airy_core PUBLIC include)
ament_target_dependencies(dual_airy_core sensor_msgs)
target_link_libraries(dual_airy_core yaml-cpp nlohmann_json::nlohmann_json)
# 常驻融合进程，与下面的启动时间检查进程顺序运行。
add_executable(dual_airy_fusion src/fusion_node.cpp)
target_link_libraries(dual_airy_fusion dual_airy_core)
ament_target_dependencies(dual_airy_fusion rclcpp sensor_msgs std_msgs geometry_msgs tf2 tf2_ros)
# 启动检查通过后退出，把会话 YAML 交给融合进程，不与点云融合同时计算。
add_executable(dual_airy_time_check src/time_check.cpp)
target_link_libraries(dual_airy_time_check dual_airy_core)
ament_target_dependencies(dual_airy_time_check rclcpp sensor_msgs std_msgs)
include(CTest)
if(BUILD_TESTING)
  add_executable(dual_airy_core_test tests/core_test.cpp)
  target_link_libraries(dual_airy_core_test dual_airy_core)
  add_test(NAME dual_airy_core_contract COMMAND dual_airy_core_test ${CMAKE_CURRENT_SOURCE_DIR}/config/extrinsics.yaml)
  add_test(NAME dual_airy_ros_environment COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/ros_env_test.sh)
endif()
install(TARGETS dual_airy_fusion dual_airy_time_check DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
