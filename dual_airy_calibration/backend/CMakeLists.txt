cmake_minimum_required(VERSION 3.16)
project(dual_airy_calibration LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
foreach(STAGE RANGE 0 2)
  if(STAGE EQUAL 0)
    set(NAME mlcc_pose)
  elseif(STAGE EQUAL 1)
    set(NAME mlcc_extrinsic)
  else()
    set(NAME mlcc_global)
  endif()
  add_executable(${NAME} src/mlcc_runner.cpp)
  target_include_directories(${NAME} PRIVATE vendor vendor/include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${NAME} ${PCL_LIBRARIES} ${OpenCV_LIBS} Eigen3::Eigen nlohmann_json::nlohmann_json Threads::Threads)
  target_compile_definitions(${NAME} PRIVATE MLCC_STAGE=${STAGE})
endforeach()
